while (hi - lo > 3) {
    int mid1 = lo + (hi - lo) / 3;
    int mid2 = hi - (hi - lo) / 3;

    int f1 = fx(mid1);
    int f2 = fx(mid2);

    if (f1 <= f2) {
        hi = mid2 - 1;
    } else {
        lo = mid1 + 1;
    }
}
